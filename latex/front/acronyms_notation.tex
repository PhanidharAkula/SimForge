% --- Symbols / Notation (non-floating, full width) ---
\noindent\textbf{Symbols}
\medskip

\setlength{\tabcolsep}{10pt}
\renewcommand{\arraystretch}{1.1}
\begin{xltabular}{\textwidth}{@{} l X @{}}
\toprule
\textbf{Symbol} & \textbf{Definition} \\
\midrule
$T_{\text{sim}}$   & Simulated time advanced by the engine (seconds) \\
$T_{\text{wall}}$  & Wall-clock runtime (seconds) \\
$\mathrm{SRT}$     & Simulated-to-real-time ratio, $\mathrm{SRT} = T_{\text{sim}} / T_{\text{wall}}$ (dimensionless) \\
$X$                & Scalar KPI (e.g., mean travel time, throughput) \\
$\mu(X)$           & Sample mean of $X$ across $N$ repeats \\
$\sigma(X)$        & Sample standard deviation of $X$ across $N$ repeats \\
$R$                & Reproducibility index, $R = 1 - \sigma(X) / \mu(X)$ (dimensionless; $R = 1$ is bit-identical) \\
$\Delta$           & Absolute difference between two scalar measurements \\
$\%\Delta$         & Relative change (percent) \\
$\Delta t$         & Discrete simulation time step (seconds) \\
$N_{\text{trips}}$ & Number of feasible trips (agents) simulated per cell \\
$|V|, |E|$         & Number of canonical-network vertices (nodes) and edges (links) \\
\bottomrule
\end{xltabular}

% --- Units ---
\noindent\textbf{Units.}\;
SI units are used throughout: length in meters (\si{m}), speed in m/s (\si{m\per s}), time in seconds (\si{s}); SLURM walltimes and benchmark runtimes are reported in (s), (min), or (h) at appropriate magnitudes. All simulation times are measured from scenario start ($t_0 = 0$).
